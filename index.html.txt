    /* ================= CART ================= */

    let cart = [];

    let orderType = "Pickup";


    /* ================= CHANGE ORDER TYPE ================= */

    function changeOrderType(type) {

      orderType = type;

      const pickupOption =
        document.getElementById("pickupOption");

      const deliveryOption =
        document.getElementById("deliveryOption");

      const pickupInfo =
        document.getElementById("pickupInfo");

      const deliveryInfo =
        document.getElementById("deliveryInfo");

      const deliveryFields =
        document.getElementById("deliveryFields");


      if (type === "Delivery") {

        pickupOption.classList.remove("active");

        deliveryOption.classList.add("active");

        pickupInfo.style.display = "none";

        deliveryInfo.style.display = "block";

        deliveryFields.classList.add("show");

      } else {

        pickupOption.classList.add("active");

        deliveryOption.classList.remove("active");

        pickupInfo.style.display = "block";

        deliveryInfo.style.display = "none";

        deliveryFields.classList.remove("show");

      }

      updateCart();

    }


    /* ================= GET DELIVERY CHARGE ================= */

    function getDeliveryCharge() {

      if (orderType === "Delivery") {
        return DELIVERY_CHARGE;
      }

      return 0;

    }


    /* ================= ADD TO CART ================= */

    function addToCart(name, price) {

      const existingItem =
        cart.find(function(item) {
          return item.name === name;
        });


      if (existingItem) {

        existingItem.quantity++;

      } else {

        cart.push({
          name: name,
          price: price,
          quantity: 1
        });

      }


      updateCart();

      openCart();

    }

        document.getElementById("cartSubtotal");

      const deliveryChargeElement =
        document.getElementById("deliveryCharge");

      const cartTotal =
        document.getElementById("cartTotal");


      cartItems.innerHTML = "";


      let subtotal = 0;

      let itemCount = 0;


      if (cart.length === 0) {

        cartItems.innerHTML = `
          <div class="cart-empty">
            Your cart is empty.
          </div>
        `;

      }


      cart.forEach(function(item, index) {

        const itemTotal =
          item.price * item.quantity;


        subtotal += itemTotal;

        itemCount += item.quantity;


        const div =
          document.createElement("div");


        div.className = "cart-item";


        div.innerHTML = `

          <div class="cart-item-top">

            <div>

              <div class="cart-item-name">
                ${item.name}
              </div>

              <div class="cart-item-price">
                ₹${itemTotal}
              </div>

            </div>


            <button
              class="remove-btn"
              onclick="removeItem(${index})">

              ❌

            </button>

          </div>


          <div class="quantity-controls">

            <button
              onclick="decreaseQuantity(${index})">

              −

            </button>


            <span class="quantity">
              ${item.quantity}
            </span>


            <button
              onclick="increaseQuantity(${index})">

              +

            </button>

          </div>

        `;


        cartItems.appendChild(div);

      });


      const deliveryCharge =
        getDeliveryCharge();


      const grandTotal =
        subtotal + deliveryCharge;


      cartCount.textContent =
        itemCount;

      cartSubtotal.textContent =
        subtotal;

      deliveryChargeElement.textContent =
        deliveryCharge;

      cartTotal.textContent =
        grandTotal;

    }


    /* ================= INCREASE QUANTITY ================= */

    function increaseQuantity(index) {

      cart[index].quantity++;

      updateCart();

    }


    /* ================= DECREASE QUANTITY ================= */

    function decreaseQuantity(index) {

      if (cart[index].quantity > 1) {

        cart[index].quantity--;

      } else {

        cart.splice(index, 1);

      }

      updateCart();

    }


    /* ================= REMOVE ITEM ================= */

    function removeItem(index) {

      cart.splice(index, 1);

      updateCart();

    }

      document
        .getElementById("cart")
        .classList.add("open");

    }


    /* ================= CLOSE CART ================= */

    function closeCart() {

      document
        .getElementById("cart")
        .classList.remove("open");

    }

        });

    /* ================= PLACE ORDER ================= */

    async function placeOrder() {

      const name =
        document
          .getElementById("customerName")
          .value
          .trim();


      const phone =
        document
          .getElementById("phone")
          .value
          .trim();


      const pickupTime =
        document
          .getElementById("pickupTime")
          .value;


      const deliveryAddress =
        document
          .getElementById("deliveryAddress")
          .value
          .trim();


      /* CART VALIDATION */

      if (cart.length === 0) {

        alert(
          "Please add food to your cart first."
        );

        openCart();

        return;

      }


      /* NAME VALIDATION */

      if (!name) {

        alert(
          "Please enter your name."
        );

        document
          .getElementById("customerName")
          .focus();

        return;

      }


      /* PHONE VALIDATION */

      if (!phone) {

        alert(
          "Please enter your mobile number."
        );

        document
          .getElementById("phone")
          .focus();

        return;

      }


      const cleanPhone =
        phone.replace(/\D/g, "");


      if (cleanPhone.length !== 10) {

        alert(
          "Please enter a valid 10-digit mobile number."
        );

        document
          .getElementById("phone")
          .focus();

        return;

      }


      /* PICKUP TIME VALIDATION */

      if (!pickupTime) {

        alert(
          "Please select your preferred time."
        );

        document
          .getElementById("pickupTime")
          .focus();

        return;

      }


      /* DELIVERY ADDRESS VALIDATION */

      if (
        orderType === "Delivery" &&
        !deliveryAddress
      ) {

        alert(
          "Please enter your delivery address."
        );

        document
          .getElementById("deliveryAddress")
          .focus();

        return;

      }


      /* CALCULATE SUBTOTAL */

      const subtotal =
        cart.reduce(
          function(sum, item) {

            return (
              sum +
              (item.price * item.quantity)
            );

          },
          0
        );


      /* DELIVERY CHARGE */

      const deliveryCharge =
        getDeliveryCharge();


      /* GRAND TOTAL */

      const total =
        subtotal + deliveryCharge;


      /* PREPARE ITEMS */

      const items =
        cart.map(function(item) {

          return {
            name: item.name,
            price: item.price,
            quantity: item.quantity
          };

        });


      /* ORDER DATA */

      const orderData = {

        customerName: name,

        phone: cleanPhone,

        orderType: orderType,

        pickupTime: pickupTime,

        deliveryAddress:
          orderType === "Delivery"
            ? deliveryAddress
            : "",

        pickupLocation:
          PICKUP_LOCATION,

        deliveryRadiusKm:
          DELIVERY_RADIUS_KM,

        subtotal: subtotal,

        deliveryCharge:
          deliveryCharge,

        total: total,

        items: items

      };


      /* BUTTON LOADING */

      const orderButton =
        document.getElementById(
          "placeOrderButton"
        );


      orderButton.disabled = true;

      orderButton.classList.add("loading");

      orderButton.textContent =
        "Placing Order...";


      /* SEND TO GOOGLE APPS SCRIPT */

      try {

        const response =
          await fetch(
            BACKEND_URL,
            {
              method: "POST",

              headers: {
                "Content-Type":
                  "text/plain;charset=utf-8"
              },

              body:
                JSON.stringify(orderData)
            }
          );


        const result =
          await response.json();


        /* SUCCESS */

        if (result.success) {

          alert(

            "🎉 Order Confirmed!\n\n" +

            "Order ID: " +
            result.orderId +

            "\n\nCustomer: " +
            name +

            "\nOrder Type: " +
            orderType +

            (
              orderType === "Delivery"
                ? "\nDelivery Charge: ₹25"
                : "\nDelivery Charge: ₹0"
            ) +

            "\nPickup/Delivery Time: " +
            pickupTime +

            "\nTotal: ₹" +
            total

          );


          /* CLEAR CART */

          cart = [];

          updateCart();


          /* CLEAR FORM */

          document
            .getElementById("customerName")
            .value = "";

          document
            .getElementById("phone")
            .value = "";

          document
            .getElementById("deliveryAddress")
            .value = "";

          document
            .getElementById("pickupTime")
            .value = "";


          closeCart();


        } else {

          alert(

            "❌ Order failed.\n\n" +

            (
              result.message ||
              "Please try again."
            )

          );

        }


      } catch (error) {

        console.error(
          "Order error:",
          error
        );


        alert(

          "❌ Unable to connect to the order server.\n\n" +

          "Please check your internet connection " +
          "and try again."

        );

      }


      /* ENABLE BUTTON */

      orderButton.disabled = false;

      orderButton.classList.remove("loading");

      orderButton.textContent =
        "Place Order";

    }


    /* ================= INITIAL CART ================= */

    updateCart();

  </script>
